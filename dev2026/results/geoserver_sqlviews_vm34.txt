grd025_annual_ts/TS_annual_grid025/featuretype.xml
WITH parameter_view AS (&#xd;
  SELECT grd025_annual_ts.*,&#xd;
    &apos;%parameter%&apos; AS parameter, &#xd;
    CASE&#xd;
      WHEN &apos;%parameter%&apos; = &apos;salinity&apos; THEN salinity&#xd;
      ELSE temperature&#xd;
    END AS value&#xd;
  FROM grd025_annual_ts&#xd;
  WHERE depth = COALESCE(%depth%, 0)&#xd;
)&#xd;
SELECT&#xd;
  parameter_view.*,&#xd;
  MIN(value) OVER () AS minvalue,&#xd;
  MAX(value) OVER () AS maxvalue&#xd;
FROM parameter_view&#xd;
WHERE value IS NOT NULL&#xd;
  AND value != &apos;NaN&apos;
   param depth          default=0          regexp=^[0-9]+$
   param parameter      default=temperature regexp=^[A-Za-z_]+$
